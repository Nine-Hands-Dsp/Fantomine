cmake_minimum_required(VERSION 3.22)
project(FantomineRender)
set(JUCE_LOCAL_PATH /home/claude/JUCE)
add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../.. fanto)
add_executable(render main.cpp)
target_link_libraries(render PRIVATE Fantomine)
target_include_directories(render PRIVATE ${CMAKE_CURRENT_LIST_DIR}/../.. $<TARGET_PROPERTY:Fantomine,INCLUDE_DIRECTORIES>)
target_compile_definitions(render PRIVATE $<TARGET_PROPERTY:Fantomine,COMPILE_DEFINITIONS>)
set_target_properties(render PROPERTIES CXX_STANDARD 17)
# conteneur de rendu sans en-tetes Xrandr / Xinerama / Xcursor (rendu hors ecran seulement)
target_compile_definitions(Fantomine PUBLIC JUCE_USE_XRANDR=0 JUCE_USE_XINERAMA=0 JUCE_USE_XCURSOR=0 JUCE_USE_XINPUT=0 JUCE_ALSA=0 JUCE_JACK=0 JUCE_MODAL_LOOPS_PERMITTED=1)
